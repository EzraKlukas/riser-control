cmake_minimum_required(VERSION 3.16)
project(riser_hardware)
find_package(ament_cmake REQUIRED)
find_package(riser_kinematics REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_dependencies(riser_kinematics)
option(RISER_WITH_IGH "Build the G5 CSV IgH backend" OFF)
if(RISER_WITH_IGH)
  find_path(IGH_INCLUDE_DIR ecrt.h HINTS /opt/etherlab/include REQUIRED)
  find_library(IGH_LIBRARY ethercat HINTS /opt/etherlab/lib REQUIRED)
  add_library(riser_bottom_csv SHARED src/bottom_csv_system.cpp)
  target_include_directories(riser_bottom_csv PRIVATE include ${IGH_INCLUDE_DIR})
  target_compile_features(riser_bottom_csv PUBLIC cxx_std_17)
  target_compile_options(riser_bottom_csv PRIVATE -Wall -Wextra -Wpedantic)
  ament_target_dependencies(riser_bottom_csv hardware_interface pluginlib rclcpp rclcpp_lifecycle)
  target_link_libraries(riser_bottom_csv ${IGH_LIBRARY} riser_kinematics::riser_kinematics)
  get_filename_component(IGH_LIBRARY_DIR "${IGH_LIBRARY}" DIRECTORY)
  set_target_properties(riser_bottom_csv PROPERTIES INSTALL_RPATH "${IGH_LIBRARY_DIR}")
  pluginlib_export_plugin_description_file(hardware_interface bottom_csv_plugins.xml)
  install(TARGETS riser_bottom_csv LIBRARY DESTINATION lib)
endif()
if(BUILD_TESTING)
  enable_testing()
  if(RISER_WITH_IGH)
    add_executable(g5_system_test test/g5_system_test.cpp)
    target_include_directories(g5_system_test PRIVATE include ${IGH_INCLUDE_DIR})
    ament_target_dependencies(g5_system_test hardware_interface pluginlib rclcpp rclcpp_lifecycle)
    target_link_libraries(g5_system_test riser_kinematics::riser_kinematics)
    add_test(NAME g5_system COMMAND g5_system_test)
    add_executable(g5_pdo_test test/g5_pdo_test.cpp)
    target_include_directories(g5_pdo_test PRIVATE include ${IGH_INCLUDE_DIR})
    ament_target_dependencies(g5_pdo_test hardware_interface rclcpp)
    add_test(NAME g5_pdo COMMAND g5_pdo_test)
  endif()
  add_executable(csv_logic_test test/csv_logic_test.cpp)
  target_include_directories(csv_logic_test PRIVATE include)
  target_compile_features(csv_logic_test PRIVATE cxx_std_17)
  add_test(NAME csv_logic COMMAND csv_logic_test)
endif()
ament_package()
