cmake_minimum_required(VERSION 3.16)
project(riser_motion_bringup)
find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(realtime_tools REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/Jog.srv" DEPENDENCIES builtin_interfaces)
rosidl_get_typesupport_target(jog_types ${PROJECT_NAME} rosidl_typesupport_cpp)
ament_export_dependencies(rosidl_default_runtime)
add_library(motion_controller SHARED src/motion_controller.cpp)
target_include_directories(motion_controller PRIVATE include)
target_compile_features(motion_controller PUBLIC cxx_std_17)
target_compile_options(motion_controller PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(motion_controller "${jog_types}")
ament_target_dependencies(motion_controller controller_interface pluginlib realtime_tools)
pluginlib_export_plugin_description_file(controller_interface plugins.xml)
install(TARGETS motion_controller LIBRARY DESTINATION lib)
install(PROGRAMS scripts/g5_console scripts/g5_jog scripts/g5_trial scripts/g5_plot_trial scripts/g5_upload scripts/g5_rebuild DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(jog_test test/jog_test.cpp)
  target_include_directories(jog_test PRIVATE include)
  target_compile_features(jog_test PRIVATE cxx_std_17)
  add_test(NAME jog_interlocks COMMAND jog_test)
  add_executable(profile_test test/profile_test.cpp)
  target_include_directories(profile_test PRIVATE include)
  target_compile_features(profile_test PRIVATE cxx_std_17)
  add_test(NAME profile COMMAND profile_test)
  add_executable(controller_test test/controller_test.cpp)
  target_include_directories(controller_test PRIVATE include)
  target_compile_features(controller_test PRIVATE cxx_std_17)
  target_link_libraries(controller_test "${jog_types}")
  ament_target_dependencies(controller_test controller_interface pluginlib realtime_tools)
  add_test(NAME controller_lifecycle COMMAND controller_test)
endif()
ament_package()
