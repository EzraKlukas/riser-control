cmake_minimum_required(VERSION 3.16)
project(riser_motion_bringup)
find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(pluginlib REQUIRED)
add_library(motion_controller SHARED src/motion_controller.cpp)
target_include_directories(motion_controller PRIVATE include)
target_compile_features(motion_controller PUBLIC cxx_std_17)
target_compile_options(motion_controller PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(motion_controller controller_interface pluginlib)
pluginlib_export_plugin_description_file(controller_interface plugins.xml)
install(TARGETS motion_controller LIBRARY DESTINATION lib)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(profile_test test/profile_test.cpp)
  target_include_directories(profile_test PRIVATE include)
  target_compile_features(profile_test PRIVATE cxx_std_17)
  add_test(NAME profile COMMAND profile_test)
  add_executable(controller_test test/controller_test.cpp)
  target_include_directories(controller_test PRIVATE include)
  target_compile_features(controller_test PRIVATE cxx_std_17)
  ament_target_dependencies(controller_test controller_interface pluginlib)
  add_test(NAME controller_lifecycle COMMAND controller_test)
endif()
ament_package()
