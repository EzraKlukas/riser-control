cmake_minimum_required(VERSION 3.16)
project(riser_gazebo)
find_package(ament_cmake REQUIRED)
find_package(riser_kinematics REQUIRED)
find_package(gz_ros2_control REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
add_library(riser_gazebo_system SHARED src/riser_system.cpp)
target_include_directories(riser_gazebo_system PRIVATE include)
target_compile_features(riser_gazebo_system PUBLIC cxx_std_17)
set_target_properties(riser_gazebo_system PROPERTIES
  CXX_STANDARD 17
  CXX_STANDARD_REQUIRED YES
  CXX_EXTENSIONS NO
)
ament_target_dependencies(riser_gazebo_system gz_ros2_control hardware_interface pluginlib rclcpp sensor_msgs)
target_link_libraries(riser_gazebo_system riser_kinematics::riser_kinematics)
target_link_libraries(riser_gazebo_system gz-sim::gz-sim)
include(CTest)
if(BUILD_TESTING)
  add_executable(force_plate_state_test test/force_plate_state_test.cpp)
  target_include_directories(force_plate_state_test PRIVATE include)
  target_compile_features(force_plate_state_test PRIVATE cxx_std_17)
  target_link_libraries(force_plate_state_test PRIVATE gz-sim::gz-sim)
  add_test(NAME force_plate_state COMMAND force_plate_state_test)
endif()
pluginlib_export_plugin_description_file(gz_ros2_control plugins.xml)
install(TARGETS riser_gazebo_system LIBRARY DESTINATION lib)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
