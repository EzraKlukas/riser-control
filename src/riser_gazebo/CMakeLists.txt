cmake_minimum_required(VERSION 3.16)
project(riser_gazebo)
find_package(ament_cmake REQUIRED)
find_package(gz_ros2_control REQUIRED)
find_package(ignition-gazebo6 REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
add_library(riser_gazebo_system SHARED src/riser_system.cpp)
target_compile_features(riser_gazebo_system PUBLIC cxx_std_17)
ament_target_dependencies(riser_gazebo_system gz_ros2_control hardware_interface pluginlib rclcpp sensor_msgs)
target_link_libraries(riser_gazebo_system ignition-gazebo6::ignition-gazebo6)
add_library(riser_loading_fixture SHARED src/loading_fixture.cpp)
target_compile_features(riser_loading_fixture PUBLIC cxx_std_17)
target_link_libraries(riser_loading_fixture ignition-gazebo6::ignition-gazebo6)
install(TARGETS riser_loading_fixture LIBRARY DESTINATION lib)
pluginlib_export_plugin_description_file(gz_ros2_control plugins.xml)
install(TARGETS riser_gazebo_system LIBRARY DESTINATION lib)
ament_package()
